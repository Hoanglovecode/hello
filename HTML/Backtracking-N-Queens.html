def is_safe(board, row, col, n):
    # Kiểm tra cột dọc
    for i in range(row):
        if board[i][col] == 1:
            return False

    # Kiểm tra đường chéo trái trên
    for i, j in zip(range(row, -1, -1), range(col, -1, -1)):
        if board[i][j] == 1:
            return False

    # Kiểm tra đường chéo phải trên
    for i, j in zip(range(row, -1, -1), range(col, n)):
        if board[i][j] == 1:
            return False

    return True

def solve_n_queens(board, row, n):
    if row >= n:
        return True # Tìm thấy lời giải

    for col in range(n):
        if is_safe(board, row, col, n):
            board[row][col] = 1 # Đặt thử
            
            if solve_n_queens(board, row + 1, n):
                return True
            
            board[row][col] = 0 # Quay lui
            
    return False

// Hàm kiểm tra vị trí (row, col) có an toàn không
bool isSafe(vector<vector<int>>& board, int row, int col, int N) {
    // Kiểm tra hàng dọc bên trên
    for (int i = 0; i < row; i++)
        if (board[i][col]) return false;

    // Kiểm tra đường chéo trái trên
    for (int i = row, j = col; i >= 0 && j >= 0; i--, j--)
        if (board[i][j]) return false;

    // Kiểm tra đường chéo phải trên
    for (int i = row, j = col; i >= 0 && j < N; i--, j++)
        if (board[i][j]) return false;

    return true;
}

// Hàm đệ quy quay lui
bool solveNQueens(vector<vector<int>>& board, int row, int N) {
    if (row >= N) return true; // Đã đặt hết N quân hậu

    for (int col = 0; col < N; col++) {
        if (isSafe(board, row, col, N)) {
            board[row][col] = 1; // Đặt quân hậu (Thử)

            if (solveNQueens(board, row + 1, N)) return true;

            board[row][col] = 0; // Quay lui (Backtrack)
        }
    }
    return false;
}

function isSafe(board, row, col, N) {
    // Kiểm tra cột dọc
    for (let i = 0; i < row; i++) {
        if (board[i][col] === 1) return false;
    }

    // Kiểm tra đường chéo trái trên
    for (let i = row, j = col; i >= 0 && j >= 0; i--, j--) {
        if (board[i][j] === 1) return false;
    }

    // Kiểm tra đường chéo phải trên
    for (let i = row, j = col; i >= 0 && j < N; i--, j++) {
        if (board[i][j] === 1) return false;
    }

    return true;
}

function solveNQueens(board, row, N) {
    if (row >= N) return true; // Đã tìm thấy lời giải

    for (let col = 0; col < N; col++) {
        if (isSafe(board, row, col, N)) {
            board[row][col] = 1; // Đặt thử

            if (solveNQueens(board, row + 1, N)) return true;

            board[row][col] = 0; // Quay lui
        }
    }
    return false;
}
