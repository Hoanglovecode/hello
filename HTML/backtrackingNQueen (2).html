# Python Backtracking
def solve(board, row, n):
    if row >= n: return True
    for col in range(n):
        if is_safe(board, row, col):
            board[row][col] = 1
            if solve(board, row + 1, n): return True
            board[row][col] = 0 # Backtrack
    return False
// C++ Backtracking
bool solve(vector<vector<int>>& b, int r, int n) {
    if (r >= n) return true;
    for (int c = 0; c < n; c++) {
        if (isSafe(b, r, c)) {
            b[r][c] = 1;
            if (solve(b, r + 1, n)) return true;
            b[r][c] = 0; // Backtrack
        }
    }
    return false;
}
