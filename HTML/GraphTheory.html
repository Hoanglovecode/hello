🔍

Kiểm Tra Euler

Đồ thị vô hướng là Euler khi và chỉ khi liên thông và mọi đỉnh đều có bậc chẵn.

  • 1Tính bậc deg(v) của từng đỉnh
  • 2Kiểm tra đồ thị liên thông (BFS)
  • 3Kiểm tra tất cả bậc chẵn
  • 4Kết luận Euler hay không
📍

Dijkstra

Tìm đường đi ngắn nhất từ đỉnh x đến y trong đồ thị có trọng số không âm.

  • 1Khởi tạo d[v] = a[x][v], T = V \ {x}
  • 2Tìm u ∈ T có d[u] nhỏ nhất
  • 3Loại u khỏi T, cố định nhãn
  • 4Cập nhật d[v] = min(d[v], d[u]+a[u][v])
🌿

Prim MST

Tìm cây phủ nhỏ nhất bằng cách mở rộng tập S bởi các cạnh có trọng số nhỏ nhất.

  • 1Thêm đỉnh 1 vào tập S đã đánh dấu
  • 2Tìm canh (i,j) nhỏ nhất: i∈S, j∉S
  • 3Thêm j vào S, ghi canh vào cây
  • 4Lặp đến khi S có đủ n đỉnh
⚡

Kruskal MST

Sắp xếp cạnh theo trọng số, dùng Union-Find để chọn cạnh không tạo chu trình.

  • 1Sắp xếp cạnh theo trọng số tăng dần
  • 2Khởi tạo Union-Find: mỗi đỉnh 1 cây
  • 3Với mỗi cạnh (u,v): nếu khác cây → thêm
  • 4Dừng khi MST có n-1 cạnh
↕

BFS Spanning Tree

Duyệt theo chiều ngang dùng hàng đợi, tạo cây phủ theo thứ tự BFS.

  • 1Đưa đỉnh đầu vào queue, đánh dấu
  • 2Lấy u ra khỏi queue
  • 3Thêm cạnh (u,v) với v chưa thăm vào cây
  • 4Đưa v vào queue, đánh dấu
↳

DFS Spanning Tree

Duyệt theo chiều sâu dùng đệ quy, tạo cây phủ theo thứ tự DFS.

  • 1Đánh dấu đỉnh u đã thăm
  • 2Với mỗi v kề u chưa thăm
  • 3Thêm cạnh (u,v) vào cây
  • 4Gọi đệ quy DFS(v)

/* De 806: Bai toan tren do thi - Euler, Dijkstra, Prim */
#include <stdio.h>
#include <stdlib.h>
#include <time.h>
#define MAX 100
#define INF 1000000

// Phat sinh do thi lien thong ngau nhien
void sinh_ma_tran(int n, int a[][MAX]) {
    // Tao cay khung de dam bao lien thong
    int thutu[MAX];
    for (int i=1; i<=n; i++) thutu[i]=i;
    // Xao tron thu tu dinh
    for (int i=n; i>=2; i--) {
        int j = rand()%i+1;
        int tmp=thutu[i]; thutu[i]=thutu[j]; thutu[j]=tmp;
    }
    for (int i=2; i<=n; i++) {
        int u=thutu[i], v=thutu[rand()%(i-1)+1];
        int w=rand()%20+1;
        a[u][v]=w; a[v][u]=w;
    }
}

// a) Kiem tra Euler: lien thong + tat ca bac chan
void kiem_tra_euler(int n, int a[][MAX]) {
    int bac[MAX]={0};
    for (int i=1;i<=n;i++)
        for (int j=1;j<=n;j++)
            if (a[i][j]>0) bac[i]++;
    int chan=1;
    for (int i=1;i<=n;i++) if (bac[i]%2!=0) {chan=0; break;}
    printf("=> Do thi %s Euler\n", chan?"LA":"KHONG PHAI");
}

// b) Dijkstra tim duong ngan nhat x -> y
void dijkstra(int n, int a[][MAX], int x, int y) {
    int d[MAX], prev[MAX], T[MAX];
    for (int i=1;i<=n;i++) {
        d[i]=a[x][i]>0?a[x][i]:INF;
        prev[i]=x; T[i]=1;
    }
    d[x]=0; T[x]=0;
    for (int iter=0;iter<n-1;iter++) {
        int u=-1;
        for (int i=1;i<=n;i++)
            if (T[i]&&(u==-1||d[i]<d[u])) u=i;
        if (u==-1||d[u]==INF) break;
        T[u]=0;
        for (int v=1;v<=n;v++)
            if (T[v]&&a[u][v]>0&&d[u]+a[u][v]<d[v]) {
                d[v]=d[u]+a[u][v]; prev[v]=u;
            }
    }
    printf("Do dai ngan nhat = %d\n", d[y]);
}
/* De 807: Cay phu theo chieu ngang - BFS Spanning Tree */
#include <stdio.h>
#define MAX 100

void bfs_span(int n, int a[][MAX], int s) {
    int tham[MAX]={0}, q[MAX*MAX], t=0, c=0;
    q[c++]=s; tham[s]=1;
    printf("Cay phu BFS tu dinh %d:\n", s);
    while (t < c) {
        int u = q[t++];
        for (int v=1; v<=n; v++)
            if (a[u][v] && !tham[v]) {
                tham[v]=1;
                printf("  %d - %d\n", u, v);
                q[c++]=v;
            }
    }
}

int main() {
    int n, a[MAX][MAX];
    printf("So dinh: "); scanf("%d", &n);
    /* Nhap ma tran ke */
    for(int i=1;i<=n;i++)
        for(int j=1;j<=n;j++)
            scanf("%d", &a[i][j]);
    int s; printf("Dinh bat dau: "); scanf("%d", &s);
    bfs_span(n, a, s);
    return 0;
}
/* De 808: DFS Spanning Tree + Prim MST */
#include <stdio.h>
#define MAX 100
#define INF 1000000

int tham[MAX];

void DFS(int n, int a[][MAX], int u) {
    tham[u]=1;
    for (int v=1; v<=n; v++)
        if (a[u][v]>0 && !tham[v]) {
            printf("  %d - %d\n", u, v);
            DFS(n, a, v);
        }
}

void prim(int n, int a[][MAX]) {
    int vis[MAX]={0}; int tong=0;
    vis[1]=1;
    for (int k=1; k<n; k++) {
        int min=INF, u=-1, v=-1;
        for (int i=1;i<=n;i++) if (vis[i])
            for (int j=1;j<=n;j++) if (!vis[j]&&a[i][j]>0&&a[i][j]<min) {
                min=a[i][j]; u=i; v=j;
            }
        if (u==-1) break;
        printf("  %d-%d: %d\n", u, v, a[u][v]);
        tong+=a[u][v]; vis[v]=1;
    }
    printf("Tong = %d\n", tong);
}
/* De 809: Kruskal MST voi Union-Find */
#include <stdio.h>
#include <stdlib.h>
#define MAX 1001

typedef struct { int u, v, w; } Edge;
int cha[MAX], rnk[MAX];

// Union-Find voi path compression
int find(int x) {
    return cha[x]==x ? x : (cha[x]=find(cha[x]));
}
int unite(int x, int y) {
    int rx=find(x), ry=find(y);
    if (rx==ry) return 0;
    if (rnk[rx]<rnk[ry]) { int t=rx; rx=ry; ry=t; }
    cha[ry]=rx;
    if (rnk[rx]==rnk[ry]) rnk[rx]++;
    return 1;
}

void kruskal(int n, Edge* e, int m) {
    for(int i=1;i<=n;i++) cha[i]=i, rnk[i]=0;
    qsort(e, m, sizeof(Edge), cmp);
    int tong=0, dem=0;
    for(int i=0; i<m && dem<n-1; i++)
        if(unite(e[i].u, e[i].v)) {
            printf("  %d-%d: %d\n", e[i].u, e[i].v, e[i].w);
            tong+=e[i].w; dem++;
        }
    printf("Tong trong so = %d\n", tong);
}
